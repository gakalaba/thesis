\begin{algorithm}[!tb]
  \caption{Ellis\_Linear Client}
  \label{ellis-lin-clientprotocol}
  \begin{algorithmic}[1]
    \State $\mathcal{L} \gets \{...\}$ \Comment{Shard Leaders}
    \State $\textit{c} \gets$ unique client ID
    \State $\sigma[\cdot] \gets 0$ \Comment{Per-shard counters}
    \State $\textit{P} \gets \textit{NULL}$
    \State $n_P \gets 0$ \Comment{Count of outstanding requests}
    \State $R : (\mathsf{Rid}\times\mathsf{Shard}) \rightharpoonup \mathsf{value};\quad R \gets \emptyset$ \Comment{Cached request results}
    \Procedure{\clientSubmit}{\textit{*Op*}, K, V}
      \State $\ell_i \in \mathcal{L} \gets shard\_leader(K)$
      \State $r \gets (c \ll 32) + \sigma[i]$ \Comment{Unique request ID}
      \State $\sigma[i] \mathrel{+}= 1$
      \If{$\textit{P} \neq \textit{NULL}$}
        \State $(r', \ell_{i'}) \gets P$ \Comment{Predecessor}
        \State $m \gets (r', r, \ell_i)$
        \State \textbf{send} $\clientToLeaderCR(m)$ to $\ell_{i'} \in \mathcal{L}$
        \EndIf
      \State $\textit{m} \gets (\textit{Op}, K, V, \textit{r}, \textit{P})$
      \State \textbf{send} $\clientToLeader(m)$ to $\ell_{i} \in \mathcal{L}$
      \State $\textit{P} \gets (r, \ell_i)$
      \State $n_P \mathrel{+}=1$
      \State \textbf{return} (r, $\ell_i$) \Comment{Return handle}
      \EndProcedure
    \Procedure{\clientWait}{r, $\ell_i$}
      \If{$(r, \ell_i) \in \mathrm{dom}(R)$}
        \State \textbf{return} $R(r,\ell_i)$
        \EndIf
      \While{\textbf{true}}
        \State \textbf{wait} receive $RequestReply(m', v)$
        \State $n_P \mathrel{-}= 1$
        \If{$n_P = 0$}
          \State $P \gets NULL$\Comment{Remove dependency}
          \EndIf
        \If{$(m'.r, m'.\ell_i) = (r, \ell_i)$}
          \State \textbf{return} $v$
          \Else
          \State $R \gets R \cup \{(m'.r, m'.\ell_i) \mapsto v\}$ \Comment{Cache result}
          \EndIf
        \EndWhile
      \EndProcedure
  \end{algorithmic}
\end{algorithm}

\begin{algorithm}[!tb]
  \caption{Ellis\_Linear Leader}
  \label{ellis-lin-leaderprotocol}
  \begin{algorithmic}[1]
    \State $\mathcal{L} \gets \{...\}$ \Comment{Shard Leaders}
    \State $\mathcal{R}$ \Comment{Set of all replicas in shard}
    \State $ShardTS \gets 0$

      \State $entry := \createntry(\textit{Op}, K, V, P, s, pred)$
      \If{$entry.\mathrm{pred} = NULL$}
        \State $entry.\mathrm{ts\_chain} = []$
        \State $my\_ts = ShardTS$
        \State $Shard\_TS = my\_ts + 1$
        \State $entry.\mathrm{ts\_chain}\append(my\_ts)$
        \State $entry.\mathrm{committed} = False$
        \State $\textit{orderedLog}\append(\textit{entry})$
        \State \textbf{send} $\textit{FinalAccept}(\textit{entry})$ to all $r \in \mathcal{R}$
        \Else
        \State $\textit{bufferedMap}\append(entry)$
        \State \textbf{send} $\textit{Accept}(entry)$ to all $r \in \mathcal{R}$
        \EndIf
    % \If{$ClientSeqnoMap[P] \neq s$}{
    %     $buffer\_{P}(\{Op, K, V, P, s\})$\\\\
    %     \algorithmiccomment{Buffer per-client out of order requests}\\\\
    %     \Return\\\\
    % }
    % \algorithmiccomment{Issue this op}\\\\
    % $\textit{bufferedMap}\append(\textit{Op}, K, V, P, s, 0)$\\\\
    % $ClientSeqnoMap[P]\mathrel{+}=1$\\\\
    % \Send $\textit{Accept}(\textit{Op}, K, V, P, s)$ to all $r \in \mathcal{R}$\\\\
    % \For{$(Op, K, V, P, s) \in buffer\_{P}$}{
    % \algorithmiccomment{Issue OoO buffered ops that are now ready}\\\\
    %     \If{$ClientSeqnoMap[P] = s$}{
    %         $bufferedMap\append{(Op, K, V, P, s, 0)}$\\\\
    %         $ClientSeqnoMap[P]\mathrel{+}=1$\\\\
    %         \Send $Accept(Op, K, V, P, s)$ to all $r \in \mathcal{R}$\\\\
    %     }
    % }
      \EndProcedure
    \Procedure{\leaderHandleAccept}{\text{entry}}
      \State $entry.\mathrm{acks}\mathrel{+}=1$
      \If{$entry.\mathrm{acks} \ge |Q|$}
        \State $entry.\mathrm{committed} \gets \textit{True}$
        \EndIf
      \If{$entry.\mathrm{committed} \land entry.\mathrm{coordinated}$}
        \State $v \gets entry.\mathrm{execute}$
        \If{$v \neq True \land entry.\mathrm{succrq} \neq NULL$}
          \State $\textit{SubmitCRResp}(entry.\mathrm{succrq}, False, NULL)$ to $L\_{succrq} \in \mathcal{L}$
          \State \textbf{return}
          \EndIf
    % \If{$timestamp\_chain \neq NULL$}{
        \label{ts_assignment_pt1}
        \State $pred\_ts = timestamp\_chain[-1]$
        \State $\textit{my\_ts} = \max(\textit{pred\_ts}+1, \textit{ShardTS})$
        \State $entry.\mathrm{ts\_chain}\append(my\_ts)$
        \label{ts_assignment_pt2}
        \State $\textit{ShardTS} = \textit{my\_ts}+1$
    % }
    % \Else{
    %     $entry.ts\_chain = NULL$\\\\
    % }
        \If{$entry.\mathrm{succrq} \neq NULL$}
          \State $\textit{SubmitCRResp}(entry.\mathrm{succrq}, True, entry.\mathrm{ts\_chain})$ to $L\_{succrq} \in \mathcal{L}$
          \EndIf
        \State $\textit{bufferedMap}\pop(\textit{entry})$
        \State $\textit{orderedLog}\append(\textit{entry})$
        \State \textbf{send} $\textit{FinalAccept}(\textit{entry})$ to all $r \in \mathcal{R}$
        \EndIf
      \EndProcedure
    \Procedure{\leaderHandleFinalAccept}{\text{entry}}
      \State $entry.\mathrm{facks}\mathrel{+}=1$
      \If{$entry.\mathrm{facks} \ge |Q|$}
        \State $entry.\mathrm{committed} = True$
        \State \textbf{send} $\textit{Commit}(\textit{entry})$ to all $r \in \mathcal{R}$
        \State $v := \execute{entry}$
        \State $\textit{SubmitOpReply}(\textit{entry}, v)$ to $\textit{client}$
        \EndIf
      \EndProcedure
  \end{algorithmic}
\end{algorithm}

\begin{algorithm}[!tb]
    \caption{Ellis\_CT Client}
    \label{ellis-ct-clientprotocol}
    \begin{algorithmic}[1]
      \State $\mathcal{L}\gets\{\ldots\}$ \Comment{Shard leaders}
      \State $c\gets$ unique client ID
      \State $\sigma[\cdot]\gets 0$ \Comment{Per-shard request counters}
  
      \State $D\gets\emptyset$
        \Comment{Invocation-Ordered dependency refcounts}
  
      \State $R\gets\emptyset$
        \Comment{Response cache}
  
      \Procedure{\clientSubmit}{\textit{Op,K,V}}
        \State $i\gets\mathsf{shard}(K)$
        \State $\ell_i\in\mathcal{L}
          \gets\mathsf{shard\_leader}(i)$
  
        \State $\rho\gets(c\ll 32)+\sigma[i]$
          \Comment{Per-shard request ID}
        \State $\sigma[i]\mathrel{+}=1$
  
        \State $P\gets\langle\mathsf{keys}(D)\rangle$
            \Comment{Invocation-Ordered predecessors}
  
        \State $m_{\mathrm{req}}
          \gets(\textit{Op},K,V,\rho,i,P)$
        \State \textbf{send} $\clientToLeader(m_{\mathrm{req}})$
          to $\ell_i\in\mathcal{L}$

        \State $h_s\gets(\rho,i)$
        % \Comment{This request is a successor to all its predecessors}
        \For{$\mathit{predIdx}\gets 0$
        \textbf{ to } $|P|-1$}
        \State $h_p\gets P[\mathit{predIdx}]$
        \State \textbf{send}
            $\clientToLeaderCR(h_p,h_s,\mathit{predIdx})$
            to $\ell_{h_p.i}$
        \EndFor
  
        \State $D(\rho,i)\gets 0$
          \Comment{Insert as future dependency}
        \State $\forall x\in\mathrm{dom}(D):
          D(x)\mathrel{+}=1$
          \Comment{Add references held by the new request}
  
        \State \textbf{return} $(\rho,i)$
      \EndProcedure
  
      \Procedure{\clientWait}{$\rho,i$}
        \If{$(\rho,i)\in\mathrm{dom}(\mathit{R})$}
          \State \textbf{return} $\mathit{R}(\rho,i)$
        \EndIf
  
        \While{\textbf{true}}
            \State \textbf{receive}
              $\RequestReply(m_{\mathrm{resp}},v)$
              \Comment{Blocking}

          \State $P^+\gets
            m_{\mathrm{resp}}.P
            \mathbin{\|}
            \langle
            (m_{\mathrm{resp}}.\rho,m_{\mathrm{resp}}.i)
            \rangle$
            \Comment{References released by response}

          \For{$\mathrm{each}\ (\rho',i')\in P^+$}
            \State $D(\rho',i')\mathrel{-}=1$
            \If{$D(\rho',i')=0$}
              \State $D\gets D\setminus\{(\rho',i')\}$
                \Comment{Remove unreferenced dependency}
            \EndIf
          \EndFor
  
          \If{$(m_{\mathrm{resp}}.\rho,m_{\mathrm{resp}}.i)
            =(\rho,i)$}
            \State \textbf{return} $v$
          \Else
            \State $\mathit{R}
              \gets\mathit{R}\cup
              \{(m_{\mathrm{resp}}.\rho,m_{\mathrm{resp}}.i)
              \mapsto v\}$
              \Comment{Cache response value}
          \EndIf
        \EndWhile
      \EndProcedure
    \end{algorithmic}
  \end{algorithm}

\begin{algorithm}[!tb]
    \caption{Ellis\_CT Shard Leader: Request Arrival}
    \label{ellis-ct-leader-timestamps}
    \begin{algorithmic}[1]
      \State $i\gets$ this shard
      \State $\ell_i\gets$ this shard's leader
      \State $\mathcal{R}_i\gets\{\ldots\}$
        \Comment{Replicas of shard $i$}
      \State $\mathit{shardTS}\gets 0$
      \State $\mathit{lastReadyTS}[\cdot]\gets 0$
        \Comment{Per-key timestamp from latest READY}
      \State $E:\mathsf{Rid}\rightharpoonup\mathsf{Entry};
        \quad E\gets\emptyset$
        \Comment{Map of operations at this shard}
      \State $\mathit{POQ}[\cdot]\gets\emptyset$
        \Comment{\parbox[t]{0.52\linewidth}{%
    For each key $K$, $\mathit{POQ}[K]$ is a
    pending-operation subqueue ordered lexicographically
    by $(\mathit{finalTS},\rho)$.}}
      \State $C\gets\langle\rangle$
        \Comment{Shard-wide commit log}
      \State $\mathit{nextExec}\gets 0$
        \Comment{Next commit-log index to execute}

      \Procedure{\leaderSubmit}{$m_{\mathrm{req}}$}
        \State $(\textit{Op},K,V,\rho,i_{\mathrm{req}},P)\gets m_{\mathrm{req}}$
        \State \textbf{assert} $i_{\mathrm{req}}=i$
        \State $e\gets\mathsf{newEntry}(m_{\mathrm{req}})$
          \Comment{Copies request fields into the entry}
        \State $e.A[0\ldots |P|-1]\gets\bot$
          \Comment{Ordered predecessor arrival timestamps}
        \State $e.F[0\ldots |P|-1]\gets\mathsf{false}$
          \Comment{Predecessor final acknowledgments}
        \State $e.S\gets\emptyset$
          \Comment{Registered successors}
        \State $e.\mathit{vr1Done}\gets\mathsf{false}$
        \State $e.\mathit{ready}\gets\mathsf{false}$
        \State $e.\mathit{vr2Done}\gets\mathsf{false}$
        \State $e.\mathit{logIdx}\gets\bot$
        \State $E(\rho)\gets e$
        \State \textbf{replicate} $m_{\mathrm{req}}$ to $\mathcal{R}_i$
          \Comment{Begin VR round 1}
      \EndProcedure
  
      \Procedure{\leaderVROneQuorum}{$\rho$}
        \State $e\gets E(\rho)$
        \If{$e.\mathit{vr1Done}$}
          \State \textbf{return}
        \EndIf
        \State $K\gets e.m_{\mathrm{req}}.K$
        \State $e.\mathit{arrivalTS}\gets
          \max(\mathit{shardTS},\mathit{lastReadyTS}[K])$
        \State $e.\mathit{finalTS}\gets e.\mathit{arrivalTS}$
        \State $\mathit{shardTS}\mathrel{+}=1$
        \State $\mathit{POQ}[K].\mathsf{insertSorted}(e)$
          \Comment{Order by $(\mathit{finalTS},\rho)$}
        \State $e.\mathit{vr1Done}\gets\mathsf{true}$
        \State $h_p\gets(e.\rho,i)$
        \For{$\mathrm{each}\;(h_s,\mathit{predIdx})\in e.S$}
          \State \textbf{send} $\ArrivalTSResponse(h_p, h_s,\mathit{predIdx}, e.\mathit{arrivalTS})$
            to $\ell_{h_s.i}$
        \EndFor
  
        \State \Call{TryReady}{$e$}
      \EndProcedure
    \end{algorithmic}
  \end{algorithm}


\paragraph{Normal-case scope.}
The algorithms describe execution under a stable shard leader.
Leader recovery and reconstruction of intermediate protocol state are
deferred.

\paragraph{Message-ordering abstraction.}
A coordination request or response received before its referenced
operation or protocol state exists is retained until that state
becomes available.
The algorithms omit the implementation-level buffering maps.
Repeated messages are idempotent: successor registrations are stored
in sets, while arrival timestamps and final acknowledgments are stored
in predecessor-indexed slots.


  \begin{algorithm}[!tb]
    \caption{Ellis\_CT Shard Leader: Request Coordination and Commit}
    \label{ellis-ct-leader-coordination}
    \begin{algorithmic}[1]
      \Procedure{\leaderSubmitCR}{$h_p,h_s,\mathit{predIdx}$}
        \State $(\rho_p,i_p)\gets h_p$
        \State $e_p\gets E(\rho_p)$
          \Comment{Predecessor operation}
        \State $e_p.S\gets
          e_p.S\cup\{(h_s,\mathit{predIdx})\}$
  
        \If{$e_p.\mathit{vr1Done}$}
          \State \textbf{send}
            $\ArrivalTSResponse(
              h_p,h_s,\mathit{predIdx},
              e_p.\mathit{arrivalTS})$
            to $\ell_{h_s.i}$
        \EndIf
  
        \If{$e_p.\mathit{ready}$}
          \State \textbf{send}
            $\FinalAck(
              h_p,h_s,\mathit{predIdx})$
            to $\ell_{h_s.i}$
        \EndIf
      \EndProcedure
  
      \Procedure{\leaderArrivalTSResp}{$h_p,h_s,\mathit{predIdx},\tau_p^{\mathrm{arr}}$}
        \State $(\rho_s,i_s)\gets h_s$
        \State $e_s\gets E(\rho_s)$
          \Comment{Successor operation}
        \State \textbf{assert} $e_s.P[\mathit{predIdx}]=h_p$
        \State $e_s.A[\mathit{predIdx}] \gets\tau_p^{\mathrm{arr}}$
          \Comment{Preserve client predecessor order}
        \State \Call{TryReady}{$e_s$}
      \EndProcedure

      \Procedure{TryReady}{$e$}
        \If{$e.\mathit{ready}$}
            \Comment{Idempotence check}
            \State \textbf{return}
        \EndIf
        \If{$\neg e.\mathit{vr1Done}\lor\bot\in e.A$}
            \State \textbf{return}
        \EndIf

        \State $K\gets e.m_{\mathrm{req}}.K$
        \State $\mathit{POQ}[K].\mathsf{remove}(e)$
            \Comment{Remove before changing its sort key}
        \State $e.\mathit{finalTS}\gets
            \max(e.\mathit{arrivalTS},\Call{\foldl}{e.A})$
        \State $\mathit{lastReadyTS}[K]\gets
            \max(\mathit{lastReadyTS}[K],
                e.\mathit{finalTS}+1)$
        \State $e.\mathit{ready}\gets\mathsf{true}$
        \State $\mathit{POQ}[K].\mathsf{insertSorted}(e)$
            \Comment{Reinsert by $(\mathit{finalTS},\rho)$}

        \State $h_p\gets(e.\rho,i)$
        \For{$\mathrm{each}\;(h_s, \mathit{predIdx})\in e.S$}
            \State \textbf{send} $\FinalAck(h_p, h_s, \mathit{predIdx})$
            to $\ell_{h_s.i}$
        \EndFor
        \State \Call{AppendReady}{$K$}
      \EndProcedure

      \Procedure{AppendReady}{$K$}
        \While{$\mathit{POQ}[K]\neq\emptyset
          \land \mathit{POQ}[K].\mathsf{head}().\mathit{ready}$}
          \State $e\gets\mathit{POQ}[K].\mathsf{popHead}()$
            \Comment{Pop the per-key POQ head}
          \State $e.\mathit{logIdx}\gets |C|$
          \State $C.\mathsf{append}(e)$
            \Comment{Append to the shard-wide commit log}
  
          \State $m_{\mathrm{commit}}\gets
            (e.\mathit{logIdx},
             e.m_{\mathrm{req}},
             e.\mathit{arrivalTS},
             e.A,
             e.\mathit{finalTS})$
          \State \textbf{replicate} $m_{\mathrm{commit}}$
            to $\mathcal{R}_i$
            \Comment{Begin VR round 2}
        \EndWhile
      \EndProcedure
  
      \Procedure{\leaderVRTwoQuorum}{$\mathit{logIdx}$}
        \State $e\gets C[\mathit{logIdx}]$
        \State $e.\mathit{vr2Done}\gets\mathsf{true}$
        \State \Call{TryExecute}{}
      \EndProcedure
    \end{algorithmic}
  \end{algorithm}


  \begin{algorithm}[!tb]
    \caption{Ellis\_CT Shard Leader: Request Execution}
    \label{ellis-ct-leader-commit-execute}
    \begin{algorithmic}[1]
      \Procedure{\leaderFinalAck}{$h_p,h_s,\mathit{predIdx}$}
        \State $(\rho_s,i_s)\gets h_s$
        \State $e_s\gets E(\rho_s)$
        \State \textbf{assert} $e_s.P[\mathit{predIdx}]=h_p$
        \State $e_s.F[\mathit{predIdx}]\gets\mathsf{true}$
        \State \Call{TryExecute}{}
      \EndProcedure
      \Procedure{TryExecute}{}
        \While{$\mathit{nextExec}<|C|$}
          \State $e\gets C[\mathit{nextExec}]$
  
          \If{$\neg e.\mathit{vr2Done}\lor\mathsf{false}\in e.F$}
            \State \textbf{return}
            \Comment{Execution barrier}
          \EndIf
  
          \State $v\gets
            \Call{Execute}{e.m_{\mathrm{req}}.\textit{Op},
                           e.m_{\mathrm{req}}.K,
                           e.m_{\mathrm{req}}.V}$
          \State \textbf{send}
            $\RequestReply(e.m_{\mathrm{req}},v)$
            to $e$'s client
          \State $\mathit{nextExec}\mathrel{+}=1$
        \EndWhile
      \EndProcedure

    \Function{\foldl}{$A$}
        \If{$|A|=0$}
            \State \textbf{return} $0$
        \EndIf
        \State $t\gets A[0]+1$
        \For{$idx\gets 1$ \textbf{to} $|A|-1$}
            \If{$A[idx]>t$}
            \State $t\gets A[idx]+1$
            \Else
            \State $t\gets t+1$
            \EndIf
        \EndFor
        \State \textbf{return} $t$
      \EndFunction
    \end{algorithmic}
  \end{algorithm}
