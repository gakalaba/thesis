\begin{algorithm}[!tb]
    \State $\mathcal{L} \gets \{...\}$ \algorithmiccomment{Shard Leaders}\\
    \State $\textit{c} \gets$ unique client ID\\
    \State $\sigma[\cdot] \gets 0$ \algorithmiccomment{Per-shard counters}\\
    \State $\textit{P} \gets \textit{NULL}$\\
    \State $n_P \gets 0$ \algorithmiccomment{Count of outstanding requests}\\
    \State $R : (\mathsf{Rid}\times\mathsf{Shard}) \rightharpoonup \mathsf{value};\quad R \gets \emptyset$ \algorithmiccomment{Cached request results}\\
    \Function{\clientSubmit(\textit{Op}, K, V)}{
        $\ell_i \in \mathcal{L} \gets shard\_leader(K)$\\
        $r \gets (c \ll 32) + \sigma[i]$ \algorithmiccomment{Unique request ID}\\
        $\sigma[i] \mathrel{+}= 1$\\
        \If{$\textit{P} \neq \textit{NULL}$}{
            $\textit{$(r', \ell_{i'})$} \gets \textit{P}$ \algorithmiccomment{Predecessor}\\
            $m \gets (r', r, \ell_i)$\\
            \Send $\clientToLeaderCR(m)$ to $\ell_{i'} \in \mathcal{L}$\\
        }
        $\textit{m} \gets (\textit{Op}, K, V, \textit{r}, \textit{P})$\\
        \Send $\clientToLeader(m)$ to $\ell_{i} \in \mathcal{L}$\\
        $\textit{P} \gets (r, \ell_i)$\\
        $n_P \mathrel{+}=1$\\
        \Return (r, $\ell_i$) \algorithmiccomment{Return handle}\\
    }
    \Function{\clientWait(r, $\ell_i$)}{
        \If{$(r, \ell_i) \in \mathrm{dom}(R)$} {
            \Return $R(r,\ell_i)$\\
        }
        \While{\textbf{true}}{
            \Wait receive $RequestReply(m', v)$\\
            $n_P \mathrel{-}= 1$\\
            \If{$n_P = 0$} {
                    $P \gets NULL$\algorithmiccomment{Remove dependency}\\
            }
            \If{$(m'.r, m'.\ell_i) = (r, \ell_i)$} {
                \Return $v$\\
            }
            \Else{
                $R \gets R \cup \{(m'.r, m'.\ell_i) \mapsto v\}$ \algorithmiccomment{Cache result}\\
            }
        }
    }
    \caption{Ellis\_Linear Client}
    \label{ellis-lin-clientprotocol}
\end{algorithm}

    \Function{\leaderHandleAccept(\text{entry})}{

             $entry.\mathrm{committed} \gets \textit{True}$\\
        }
        \If{$entry.\mathrm{committed} \land entry.\mathrm{coordinated}$}{
            $v \gets entry.\mathrm{execute}$\\
            \If{$v \neq True \land entry.\mathrm{succrq} \neq NULL$}{
                $\textit{SubmitCRResp}(entry.\mathrm{succrq}, False, NULL)$ to $L_{succrq} \in \mathcal{L}$\\
                \Return
            }
            % \If{$timestamp\_chain \neq NULL$}{

                $entry.\mathrm{ts\_chain}\append(my\_ts)$\\
                \label{ts_assignment_pt2}
                $\textit{ShardTS} = \textit{my\_ts}+1$\\
            % }
            % \Else{
            %     $entry.ts\_chain = NULL$\\
            % }
            \If{$entry.\mathrm{succrq} \neq NULL$}{
                $\textit{SubmitCRResp}(entry.\mathrm{succrq}, True, entry.\mathrm{ts\_chain})$ to $L_{succrq} \in \mathcal{L}$\\
            }
            $\textit{bufferedMap}\pop(\textit{entry})$\\
            $\textit{orderedLog}\append(\textit{entry})$\\
            \Send $\textit{FinalAccept}(\textit{entry})$ to all $r \in \mathcal{R}$\\             
        }
    }
    \Function{\leaderHandleFinalAccept(\text{entry})}{
        $entry.\mathrm{facks}\mathrel{+}=1$\\
        \If{$entry.\mathrm{facks} \ge |Q|$}{
            $entry.\mathrm{committed} = True$\\
            \Send $\textit{Commit}(\textit{entry})$ to all $r \in \mathcal{R}$\\
            $v := \execute{entry}$\\
            $\textit{SubmitOpReply}(\textit{entry}, v)$ to $\textit{client}$\\
        }
    }   
    \caption{Ellis\_Linear Leader}
    \label{ellis-lin-leaderprotocol}
\end{algorithm}

\begin{algorithm}[!tb]
    \State $\mathcal{L} \gets \{...\}$ \algorithmiccomment{Shard leaders}\\
    \State $c \gets$ unique client ID\\
    \State $\sigma[\cdot] \gets 0$ \algorithmiccomment{Per-shard counters}\\
    \State $D : (\mathsf{Rid}\times\mathsf{Shard}) \rightharpoonup \mathbb{N};\quad D \gets \emptyset$ \algorithmiccomment{Outstanding dependencies}\\
    \State $R : (\mathsf{Rid}\times\mathsf{Shard}) \rightharpoonup \mathsf{value};\quad R \gets \emptyset$ \algorithmiccomment{Cached request results}\\

    \Function{\clientSubmit(\textit{Op}, K, V)}{
        $\ell_i \in \mathcal{L} \gets shard\_leader(K)$\\
        $r \gets (c \ll 32) + \sigma[i]$ \algorithmiccomment{Unique request ID}\\
        $\sigma[i] \mathrel{+}= 1$\\
        $P \gets \mathrm{dom}(D)$ \algorithmiccomment{Predecessors}\\
        \For{$\mathrm{each}\ (r', \ell_{i'}) \in P$}{
            $m \gets (r', r, \ell_i)$\\
            \Send $\clientToLeaderCR(m)$ to $\ell_{i'} \in \mathcal{L}$\\
        }
        $m \gets (\textit{Op}, K, V, r, P)$\\
        \Send $\clientToLeader(m)$ to $\ell_{i} \in \mathcal{L}$\\
        $D(r, \ell_{i}) \gets 0$ \algorithmiccomment{Insert as future dependency}\\
        $\forall x \in \mathrm{dom}(D):\ D(x) \mathrel{+}= 1$\\
        \Return (r, $\ell_i$) \algorithmiccomment{Return handle}\\
    }

    \Function{\clientWait(r, $\ell_i$)}{
        \If{$(r, \ell_i) \in \mathrm{dom}(R)$} {
            \Return $R(r,\ell_i)$\\
        }
        \While{\textbf{true}}{
            \Wait receive $RequestReply(m', v)$\\
            \For{$\mathrm{each}\ (r', \ell_{i'}) \in m'.P$}{
                $D(r', \ell_{i'}) \mathrel{-}= 1$\\
                \If{$D(r', \ell_{i'}) = 0$} {
                    $D \gets D \setminus \{(r', \ell_{i'})\}$\algorithmiccomment{Remove dependency}\\
                }
            }
            \If{$(m'.r, m'.\ell_i) = (r, \ell_i)$} {
                \Return $v$\\
            }
            \Else{
                $R \gets R \cup \{(m'.r, m'.\ell_i) \mapsto v\}$ \algorithmiccomment{Cache result}\\
            }
        }
    }

    \caption{Ellis\_CT Client}
    \label{ellis-ct-clientprotocol}
\end{algorithm}

\begin{algorithm}[!tb]
    \State $\mathcal{R}$ \algorithmiccomment{Shard replicas}\\
    \State $ShardTS \gets 0$\\
    \State $\textit{unorderedBag} \gets \emptyset$\\
    \State $\textit{orderedLog} \gets []$\\
    \Function{\leaderSubmit(Op, K, V, r, P)}{
        $entry := \createntry(Op, K, V, r, P)$\\
        $\textit{unorderedBag}\append(entry)$\\
        \Send $\Accept(entry)$ to all $r \in \mathcal{R}$ \algorithmiccomment{Unordered replication}\\
    }
    \Function{\leaderHandleAccept(\textit{entry})}{
        $entry.\mathrm{acks} \mathrel{+}= 1$\\
        \If{$entry.\mathrm{acks} \ge |Q|$}{
            $entry.\mathrm{committed} \gets True$\\
            $entry.\mathrm{arrival\_TS} \gets ShardTS$; $ShardTS \gets ShardTS + 1$\\
            $\textit{orderedLog}\insertSorted(entry)$\\
            $entry.\mathrm{ready} \gets False$\\
            \Send $\coordresp(entry.\mathrm{arrival\_TS})$ to all $\ell_i \in$ \textit{entry}.successors\\
        }
    }
    \Function{\leaderSubmitCR(\textit{pred}, \textit{req})}{
        $pred.\mathrm{succrq} := \textit{req}$\\
        \If{$entry := \textit{orderedLog}\find(pred) \neq NULL$}{
            \Send $\coordresp(entry.\mathrm{arrival\_TS})$ to $L_{req}$\\
        }
        \Else{
            \Wait $entry := \textit{unorderedBag}\find(pred) \neq NULL$\\
            \Wait $entry.\mathrm{committed}$\\
            \Send $\coordresp(entry.\mathrm{arrival\_TS})$ to $L_{req}$\\
        }
    }
    \Function{\leaderCoordResp(\textit{req}, \textit{pred\_arrivals})}{
        \Wait $entry := \textit{orderedLog}\find(req) \neq NULL$\\
        $entry.\mathrm{final\_TS} \gets \max(\foldl(pred\_arrivals), entry.\mathrm{arrival\_TS})$\\
        $entry.\mathrm{ready} \gets True$\\
        $ShardTS \gets entry.\mathrm{final\_TS} + 1$\\
        \Send $\FinalAccept(entry)$ to all $r \in \mathcal{R}$\\
    }
    \Function{\leaderFinalCoordResp(req)}{
        \Wait $tis$\\
    }
    \Function{\leaderHandleFinalAccept(\textit{entry})}{
        $entry.\mathrm{facks} \mathrel{+}= 1$\\
        \If{$entry.\mathrm{facks} \ge |Q|$}{
            \Send $\commit(entry)$ to all $r \in \mathcal{R}$\\
            $v := \execute(entry)$\\
            \Send $\requestreply(entry, v)$ to $\textit{client}$\\
        }
    }
    \caption{Ellis\_CT Leader}
    \label{ellis-ct-leaderprotocol}
\end{algorithm}

\begin{algorithm}[!tb]
    \State $\mathcal{L} \gets \{...\}$ \algorithmiccomment{Shard Leaders}\\
    \State $\mathcal{R}$ \algorithmiccomment{Set of all replicas in shard}\\
    \State $ShardTS \gets 0$\\
    \State $last\_ready\_TS[k] \gets 0, \; \forall k$ \algorithmiccomment{Per-key counter}\\

    \Function{\leaderSubmit(\textit{Op}, K, V, P, s, preds)}{
        $entry := \createntry(\textit{Op}, K, V, P, s, preds)$\\
        $entry.\mathrm{arrival\_ts} = ShardTS$\\
        $ShardTS \gets ShardTS + 1$\\
        $\textit{bufferedSet}\append(entry)$\\
        \Send $\textit{Accept}(entry)$ to all $r \in \mathcal{R}$\\
        \Send $\textit{ArrivalTS}(entry.\mathrm{arrival\_ts})$ to all successors in $\mathcal{L}$\\
    }

    \Function{\leaderHandleArrivalAcks(entry)}{
        \algorithmiccomment{All predecessor arrival timestamps received}\\
        \Wait $entry.\mathrm{pred\_arrival\_ts}[i] \neq \textit{NULL}$ for all $i \in entry.\mathrm{preds}$\\
        $folded \gets \texttt{foldl}(entry.\mathrm{pred\_arrival\_ts})$\\
        $entry.\mathrm{assigned\_ts} \gets \max(folded, last\_ready\_TS[entry.K] + 1)$\\
        $\textit{orderedLog}\insertSorted(entry, key = entry.\mathrm{assigned\_ts})$\\
        \Send $\textit{AssignedTS}(entry.\mathrm{assigned\_ts})$ to all successors in $\mathcal{L}$\\
    }

    \Function{\leaderHandleAssignedAcks(entry)}{
        \algorithmiccomment{All predecessor assigned timestamps received (2nd round)}\\
        \Wait $entry.\mathrm{pred\_assigned\_ts}[i] \neq \textit{NULL}$ for all $i \in entry.\mathrm{preds}$\\
        $entry.\mathrm{final\_ts} \gets \texttt{foldl}(entry.\mathrm{pred\_assigned\_ts})$\\
        $entry.\mathrm{ready} \gets \textit{True}$\\
        $last\_ready\_TS[entry.K] \gets entry.\mathrm{final\_ts}$\\
        $\textit{orderedLog}\resort(entry, sortkey = entry.\mathrm{final\_ts})$\\
    }

    \Function{\leaderCommitAndExecute(entry)}{
        \algorithmiccomment{Commit when entry reaches head of log and is ready}\\
        \Wait $\textit{orderedLog.head()} == entry \land entry.\mathrm{ready}$\\
        \If{\textit{no unexecuted entry on same key precedes entry}}{
            $v := \execute(entry)$\\
            $\textit{SubmitOpReply}(entry, v)$ to $entry.\mathrm{client}$\\
        }
    }

    \caption{Ellis\_CT Leader (CONTINUEEEEDD MERGE SECOND HALF)}
    \label{alg:iocl_no_failures}
\end{algorithm}



\begin{algorithm}[!tb]

        $entry.\mathrm{status} \gets \textit{FINAL}$\\
        $\textit{orderedLog}\insertSorted(entry, key = entry.\mathrm{final\_ts})$\\
        \Send $\textit{Commit}(entry)$ to all $r \in \mathcal{R}$\\
    }

    \Function{\leaderCommitAndExecute(entry)}{
        \algorithmiccomment{Replicas ack final commit}\\
        \Wait $\textit{orderedLog.head()} == entry \land entry.\mathrm{ready}$\\
        \If{\textit{no unexecuted entry on same key precedes entry}}{
            $v := \execute(entry)$\\
            $\textit{SubmitOpReply}(entry, v)$ to $entry.\mathrm{client}$\\
        }
    }

    \label{alg:iocl_fail_tolerant}
\end{algorithm}

        }
        \If{$entry.\mathrm{committed} \land entry.\mathrm{coordinated}$}{
            $entry.\mathrm{my\_sort\_ts} \gets \textit{foldl}(entry.\mathrm{pred\_ts})$\\
            $\textit{bufferedMap}\pop(\textit{entry})$\\
            $\textit{sortedTempLog}\insertSorted(\textit{entry})$\\
            $\textit{SubmitCRAckResp}(entry.\mathrm{my\_sort\_ts}, \textit{req}, entry.\mathrm{succrq})$ to $L_{succrq} \in \mathcal{L}$\\
        }    
    }
    \Function{\CRAckReply(\textit{ts}, \textit{pred}, \textit{req})}{
        \Wait $entry := sortedTempLog\find{req} \neq NULL$\\
        \Wait $sortedTempLog.hd() == entry$\\
        $\textit{sortedTempLog}\pop(\textit{entry})$\\
        $\textit{orderedLog}\append(\textit{entry})$\\
        $entry.\mathrm{my\_ts} \gets \mathrm{max}(ShardTS, entry.\mathrm{my\_sort\_ts})$\\
        $\mathrm{ShardTS} \gets entry.\mathrm{my\_ts} + 1$\\
        $\mathrm{Running\_ShardTS} \gets \mathrm{ShardTS}$\\
        \Send $\textit{FinalAccept}(\textit{entry})$ to all $r \in \mathcal{R}$\\             
    }
    \caption{IOCL Leader Coordination}
    \label{shardprotocolcoord}
\end{algorithm}
